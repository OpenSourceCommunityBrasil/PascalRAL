procedure TDados.DataModuleCreate(Sender: TObject);
{{#CRUD}}
var
  vBanco: TRALDBBase;
  vLinhas, vUltimo: Int64RAL;
{{#LAZARUS}}
  {$IFDEF LINUX}
  vLib: string;
  {$ENDIF}
{{/LAZARUS}}
{{/CRUD}}
begin
{{#CRUD}}
{{#LAZARUS}}
  {$IFDEF LINUX}
  // o sqldb procura libsqlite3.so, que so vem com o pacote de desenvolvimento:
  // a biblioteca de execucao, onde a distribuicao a poe
  if Banco.LibLocation = '' then
    for vLib in ['/usr/lib/x86_64-linux-gnu/libsqlite3.so.0', '/usr/lib/aarch64-linux-gnu/libsqlite3.so.0',
      '/usr/lib64/libsqlite3.so.0', '/usr/lib/libsqlite3.so.0'] do
      if FileExists(vLib) then
      begin
        Banco.LibLocation := vLib;
        Break;
      end;
  {$ENDIF}
{{/LAZARUS}}
  // a tabela do recurso /itens, criada na primeira vez (SQLite em dados.sqlite)
  vBanco := Banco.AcquireDatabase(nil, nil);
  try
    vBanco.ExecSQL('create table if not exists itens (id integer primary key, ' +
      'nome varchar(100) not null)', nil, vLinhas, vUltimo);
  finally
    Banco.ReleaseDatabase(vBanco);
  end;
{{/CRUD}}
end;
